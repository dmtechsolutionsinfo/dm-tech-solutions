Google Drive Backup & Apps Script Guide

1. Click Export Backup (.JSON) in Settings and upload the file to your Google Drive folder anytime.
2. Or create a Google Apps Script Web App on script.google.com using the template below to save JSON snapshots directly to your Google Drive: 

function doPost(e) {
  var folderName = "DM Tech Solutions Backups";
  var folders = DriveApp.getFoldersByName(folderName);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
  var payload = e.postData ? e.postData.contents : "{}";
  var fileName = "DM_Tech_Backup_" + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd_HH-mm") + ".json";
  var file = folder.createFile(fileName, payload, MimeType.PLAIN_TEXT);
  return ContentService.createTextOutput(JSON.stringify({ ok: true, fileId: file.getId(), name: fileName }))
    .setMimeType(ContentService.MimeType.JSON);
}
